// A fabric-covered, exposed-linkage experimental aircraft, not a bird costume.
export function createAircraft(T,p){
  const root=new T.Group();root.name='Tinn / Resonant 14';
  const a={root,wingPanels:{},wheels:[],T,p,animatedRods:[],wingDetail:[],wrists:[]};
  const material=(color,metalness=0,roughness=.65)=>new T.MeshStandardMaterial({color,metalness,roughness});
  const m=a.materials={
    fabric:new T.MeshStandardMaterial({color:0xe7d6a7,side:T.DoubleSide,roughness:.86,metalness:0}),
    outer:new T.MeshStandardMaterial({color:0xf0e4c7,side:T.DoubleSide,roughness:.8}),
    frame:material(0x284b4b,.65,.35),alloy:material(0xa9b4ad,.82,.29),
    dark:material(0x202c30,.48,.43),brass:material(0xbb8d42,.7,.32),
    red:material(0xa64630,.25,.5),rubber:material(0x181c1b,0,.93),
    leather:material(0x553a2b,0,.85),suit:material(0x546878,0,.9),
    skin:material(0xc48e6b,0,.8),visor:material(0x182e3b,.45,.23),
    white:material(0xd7d1b9,0,.75)
  };
  const mesh=(geo,mat,parent=root,pos=[0,0,0])=>{
    const obj=new T.Mesh(geo,mat);obj.position.set(...pos);parent.add(obj);return obj;
  };
  const box=(size,pos,mat,parent=root)=>mesh(new T.BoxGeometry(...size),mat,parent,pos);
  const sphere=(scale,pos,mat,parent=root)=>{
    const o=mesh(new T.SphereGeometry(1,16,10),mat,parent,pos);o.scale.set(...scale);return o;
  };
  const rod=(v,w,r,mat,parent=root)=>{
    const obj=mesh(new T.CylinderGeometry(r,r,1,8),mat,parent);
    fitRod(obj,v,w);return obj;
  };
  a.rod=rod;
  // The four wing mounting groups and their surfaces stay identity transforms.
  for(const name of PANEL_NAMES){
    const group=a[name]=new T.Group(),surface=a[name+'Surface']=new T.Group();
    group.name=name;surface.name=name+'Surface';root.add(group);group.add(surface);
    a.wingPanels[name]=[];
    const count=name.endsWith('Outer')?2:1;
    for(let leaf=0;leaf<count;leaf++){
      const panel=mesh(makePanel(T,p,name,leaf),count===2?m.outer:m.fabric,surface);
      panel.name=name+' / rigid sliding leaf '+leaf;
      a.wingPanels[name].push(panel);
    }
    // Slender spars, rib caps and perimeter tapes. These are mechanical detail,
    // not substitute surfaces. Every member follows the production linkage.
    for(let leaf=0;leaf<count;leaf++){
      const divisions=count===2?4:7;
      for(let j=0;j<=divisions;j++){
        for(let k=0;k<2;k++){
          const obj=rod([0,0,0],[0,1,0],.008,m.frame);
          a.wingDetail.push({obj,name,leaf,t0:j/divisions,t1:j/divisions,u0:k/2,u1:(k+1)/2});
        }
      }
      for(let j=0;j<divisions;j++){
        for(const u of [0,.25,1]){
          const obj=rod([0,0,0],[0,1,0],u===.25?.026:.012,u===.25?m.frame:m.brass);
          a.wingDetail.push({obj,name,leaf,t0:j/divisions,t1:(j+1)/divisions,u0:u,u1:u});
        }
      }
    }
  }
  // Triangulated fuselage: two longerons each side, narrowing into a tail boom.
  const stations=[
    {x:2.5,y:-.12,w:.18,h:.26},{x:1.5,y:-.15,w:.43,h:.4},
    {x:.35,y:-.15,w:.46,h:.48},{x:-.7,y:-.13,w:.36,h:.48},
    {x:-2.9,y:.02,w:.08,h:.13}
  ];
  for(let j=0;j<stations.length-1;j++){
    const b=stations[j],d=stations[j+1];
    for(const side of [-1,1]){
      for(const high of [-1,1]){
        rod([b.x,b.y+high*b.h,side*b.w],[d.x,d.y+high*d.h,side*d.w],j===3?.025:.031,m.frame);
      }
      rod([b.x,b.y-b.h,side*b.w],[d.x,d.y+d.h,side*d.w],.018,m.alloy);
      rod([b.x,b.y+b.h,side*b.w],[b.x,b.y-b.h,side*b.w],.022,m.frame);
    }
    rod([b.x,b.y-b.h,-b.w],[b.x,b.y-b.h,b.w],.026,m.frame);
  }
  // A compact nose shell and open cockpit; no opaque bulb concealing the pilot.
  sphere([.61,.31,.34],[2.05,-.06,0],m.red);
  box([.9,.08,.78],[1.05,-.55,0],m.dark);
  box([.14,.7,.72],[.42,-.05,0],m.leather).rotation.z=-.12;
  box([.61,.1,.68],[.78,-.38,0],m.leather);
  for(const side of [-1,1]){
    rod([1.48,.08,.44*side],[.4,.18,.47*side],.025,m.alloy);
    rod([.4,.18,.47*side],[.25,.55,.42*side],.025,m.frame);
  }
  // Engine, cooling fins, exhaust and fuel tank sit aft of the seat.
  box([.61,.45,.55],[-.56,-.12,0],m.dark);
  for(let i=0;i<9;i++)box([.39,.025,.61],[-.57,.04+i*.035,0],m.alloy);

  rod([-.55,.18,-.37],[-1.15,.13,-.42],.065,m.dark);
  rod([-1.15,.13,-.42],[-1.38,-.07,-.42],.065,m.dark);
  box([.63,.36,.58],[-1.24,-.08,0],m.red);

  cap.name='five litre tank filler';
  // Reduction gearbox, crank axle, twin spring cartridges and shoulder bearings.
  box([.45,.34,.63],[0,.42,0],m.frame);

    const disc=mesh(new T.CylinderGeometry(.23,.23,.045,24),m.brass,a.crank,[0,0,z]);
    disc.rotation.x=Math.PI/2;
    rod([0,0,z],[.18,0,z],.035,m.dark,a.crank);
    sphere([.055,.055,.055],[.18,0,z],m.alloy,a.crank);
  }
  for(const side of [-1,1]){
    rod([-.33,.05,.32*side],[-.12,.74,.25*side],.065,m.dark);
    for(let i=0;i<11;i++){
      const ring=mesh(new T.TorusGeometry(.069,.012,5,12),m.brass,root,[-.33+.21*i/11,.05+.69*i/11,.32*side-.07*side*i/11]);

    const carriage=sphere([.075,.075,.075],[0,0,0],m.alloy);

    const tie=rod([0,0,0],[0,1,0],.018,m.alloy);
    a.wrists.push({side,bearing,carriage,rail,tie});
  }
  // Visible belt drive (no propeller, jet or fictitious thrust apparatus).
  for(const z of [-.22,.22]){
    rod([-.56,-.1,z],[0,.42,z],.018,m.rubber);
    rod([-.78,-.02,z],[-.17,.5,z],.018,m.rubber);

  const tailMat=m.fabric;
  a.elevator=new T.Group();a.elevator.position.set(-2.8,.02,0);root.add(a.elevator);

    for(let k=0;k<6;k++){
      const angle=k*Math.PI/3;
      rod([0,0,.067],[.16*Math.cos(angle),.16*Math.sin(angle),.067],.009,m.alloy,tyre);
    }
    if(i===0){
      for(const z of [-.105,.105])rod([1.77,-.35,z],[g[0],g[1],z],.035,m.frame);
    }else{
      rod([.3,-.43,.29*Math.sign(g[2])],g,.035,m.frame);
      rod([-.85,-.42,.3*Math.sign(g[2])],g,.029,m.alloy);
    }
  }

  // with arms on a stick and restrained shoulders. All dimensions are metres.
  a.pilot=new T.Group();a.pilot.name='55 kg seated pilot';root.add(a.pilot);
  const pilot=a.pilot;
  sphere([.18,.22,.23],[.83,-.17,0],m.suit,pilot);
  const torso=sphere([.18,.34,.245],[.72,.22,0],m.suit,pilot);torso.rotation.z=.12;
  rod([.72,.47,0],[.75,.57,0],.075,m.skin,pilot);
  sphere([.125,.165,.12],[.79,.69,0],m.skin,pilot);
  sphere([.151,.142,.151],[.765,.77,0],m.white,pilot);
  sphere([.07,.067,.127],[.893,.735,0],m.visor,pilot);
  sphere([.035,.043,.05],[.924,.668,0],m.skin,pilot);
  for(const side of [-1,1]){
    const z=side*.13;
    rod([.89,-.21,z],[1.29,-.32,z],.095,m.suit,pilot);

    rod([1.29,-.32,z],[1.53,-.65,z],.067,m.suit,pilot);

    sphere([.067,.065,.067],[1.04,.12,.29*side],m.suit,pilot);
    rod([1.04,.12,.29*side],[1.29,.12,.08*side],.051,m.suit,pilot);

function fitRod(obj,v,w){
  // THREE's cylinder primitive is aligned with +Y.
  const dx=w[0]-v[0],dy=w[1]-v[1],dz=w[2]-v[2],length=Math.hypot(dx,dy,dz);
  obj.position.set((v[0]+w[0])/2,(v[1]+w[1])/2,(v[2]+w[2])/2);
  obj.scale.y=Math.max(1e-7,length);
  // Quaternion from [0,1,0] to direction, without allocating a vector.
  if(length>1e-10){
    const x=dx/length,y=dy/length,z=dz/length;
    if(y<-.999999)obj.quaternion.set(1,0,0,0);
    else obj.quaternion.set(z,0,-x,1+y).normalize();
  }
}

// Detail point uses the same material coordinates and two rigid leaves as the
// supplied production panel. The host remains sole owner of panel vertices.
function detailPoint(p,pose,name,leaf,t,u){
  const g=planform(p),outer=name.endsWith('Outer'),side=name.startsWith('left')?-1:1;
  const v=pose[name],inn=pose[side<0?'leftInner':'rightInner'];
  const stroke=inn.stroke+(outer?v.stroke:0);
  const mt=outer?(leaf*.5+t*.5):t;
  const k=Math.min(3,Math.floor(mt*4)),lo=k/4,hi=(k+1)/4;
  const chord=outer?g.chord*(chordFactor(lo)+(chordFactor(hi)-chordFactor(lo))*(mt-lo)/(hi-lo)):g.chord;
  const r=outer?g.outer*(leaf===0?t*.5:v.exposure-.5+t*.5):g.inner*t;
  const x=(.25-u)*chord,y=4*u*(1-u)*chord*v.camber-(outer&&leaf===1?.008:0);
  const ca=Math.cos(v.twist),sa=Math.sin(v.twist),cs=Math.cos(stroke),ss=Math.sin(stroke);
  const xx=x*ca-y*sa,yy=x*sa+y*ca;
  return [xx+v.sweep,E.wing.root_height_m+(outer?g.inner*Math.sin(inn.stroke):0)+r*ss+yy*cs,
    side*(g.hinge+(outer?g.inner*Math.cos(inn.stroke):0)+r*cs-yy*ss)];
}
export function updateAircraft(a,s,c){
  const pose=wingPose(s);
  for(const w of a.wheels)w.rotation.set(0,0,s.wheel_angle??0);
  a.rudder.rotation.y=-(c.rudder??0)*.4;
  a.elevator.rotation.z=-(c.pitch??0)*.22;
  a.crank.rotation.z=s.phase;
  for(const item of a.wingDetail){
    fitRod(item.obj,detailPoint(a.p,pose,item.name,item.leaf,item.t0,item.u0),
      detailPoint(a.p,pose,item.name,item.leaf,item.t1,item.u1));
  }
  for(const item of a.animatedRods){
    const z=item.side*.35,phi=s.phase;
    fitRod(item.obj,[.18*Math.cos(phi),.58+.18*Math.sin(phi),z],
      detailPoint(a.p,pose,item.side<0?'leftInner':'rightInner',0,.22,.25));
  }
  for(const item of a.wrists){
    const prefix=item.side<0?'left':'right';
    const inn=detailPoint(a.p,pose,prefix+'Inner',0,1,.25);
    const out=detailPoint(a.p,pose,prefix+'Outer',0,0,.25);
    item.bearing.position.set(...inn);item.carriage.position.set(...out);
    fitRod(item.rail,inn,out);
    fitRod(item.tie,detailPoint(a.p,pose,prefix+'Inner',0,.88,.65),
      detailPoint(a.p,pose,prefix+'Outer',0,.13,.65));
  }
}

import {E} from './environment/engineering.mjs';

// A cam-and-linkage completion of the readable action, not sampled notebook data.
// Timing is a fixed linkage setting; only the dynamical phase advances the cam.
export function wingPose(s){
  const w=E.wing,phi=s.gliding?0:s.phase;
  const b=s.timing_bias??w.timing_bias_low_speed;
  const theta=phi-b*Math.sin(phi),sn=Math.sin(theta),cs=Math.cos(theta);
  const scale=s.sweep_scale??1;
  const inner=s.gliding?
    {stroke:w.dihedral_rad,twist:w.incidence_rad,sweep:0,exposure:1,camber:w.camber_inner_base}:
    {stroke:w.dihedral_rad-w.inner_amplitude_rad*sn,
     twist:w.incidence_rad-w.inner_twist_amplitude_rad*cs,
     sweep:-.5*w.sweep_amplitude_m*sn*scale,exposure:1,
     camber:w.camber_inner_base+w.camber_inner_amplitude*cs};
  const outer=s.gliding?
    {stroke:0,twist:w.incidence_rad,sweep:0,exposure:1,camber:w.camber_outer_base}:
    {stroke:-w.outer_amplitude_rad*sn,
     twist:w.incidence_rad-w.outer_twist_amplitude_rad*cs,
     sweep:-w.sweep_amplitude_m*sn*scale,
     exposure:1-(1-w.recovery_exposure)*(.5-.5*Math.tanh(4*cs)/Math.tanh(4)),
     camber:w.camber_outer_base+w.camber_outer_amplitude*cs};
  return {leftInner:{...inner},rightInner:{...inner},leftOuter:{...outer},rightOuter:{...outer}};
}

import {E} from './environment/engineering.mjs';
import {wingPose} from './kinematics.mjs';
import {mechanics,add,sub,mul,dot,cross,unit,bodyToWorld,worldToBody} from './environment/source-mechanism.mjs';

const clamp=(x,a,b)=>Math.max(a,Math.min(b,x));
// Photo 02: rounded "Say" span 7.4 ft, original area paragraph 6.61 ft².
const AR=7.4**2/6.61;
export const parameters={
  ...E,aspect_ratio:AR,wrist_span_fraction:3.7/7.4,
  wing_area_m2:14**2/AR,dry_mass_kg:173.9,mass_kg:177.5,
  gear_centres:E.gear_centres_m.map(v=>v.slice())
};

export function polar(region,alpha,camber){
  const table=E.sections[region.toLowerCase()],xs=E.sections.alpha_rad;
  let i=0;
  const a=clamp(alpha,xs[0],xs[xs.length-1]);
  while(i<xs.length-2&&a>xs[i+1])i++;
  const t=(a-xs[i])/(xs[i+1]-xs[i]);
  const interp=v=>v[i]+t*(v[i+1]-v[i]);
  const CL=interp(table.cl_base)+table.camber_lift_gain*(camber-table.camber_reference)*Math.cos(alpha);
  const CD=interp(table.cd_base)+CL*CL/(Math.PI*E.aero.span_efficiency*AR);
  return {CL,CD,cl:CL,cd:CD};
}

export function forces(s,c,e,poseOverride){
  const frozenPose=typeof poseOverride==='function'?poseOverride(s):poseOverride;
  const q=mechanics(parameters,s,frozenPose?()=>frozenPose:wingPose);
  const rho=e.rho??E.rho_kg_m3,g=e.gravity??E.gravity_m_s2;
  const air=worldToBody(sub(s.velocity,e.wind??[0,0,0]),s);
  let total=[0,0,0],wing=[0,0,0],Qphi=0,Qpitch=0,loss=0,bodyDrag=0;
  const apply=(F,jphi,jtheta)=>{
    total=add(total,F);Qphi+=dot(F,jphi);Qpitch+=dot(F,jtheta);
  };
  for(const v of q.samples){
    if(v.area<=0)continue;
    let u=add(air,v.velocity);
    u=sub(u,mul(v.span,dot(u,v.span)));
    const speed=Math.hypot(...u);
    if(speed<1e-12)continue;
    const d=mul(u,1/speed);
    const alpha=Math.atan2(-dot(d,v.normal),dot(d,v.chord));
    const {CL,CD}=polar(v.region,alpha,v.camber);
    const pressure=.5*rho*speed*speed*v.area;
    const F=mul(sub(mul(unit(cross(v.span,d)),CL),mul(d,CD)),pressure);
    wing=add(wing,F);apply(F,v.jphi,v.jtheta);
    loss+=pressure*CD*speed;
  }
  const r0=mul(q.cg,-1),j0=mul(q.dc,-1);
  const local=(r,j)=>add(air,add(mul(j,s.omega),mul([-r[1],r[0],0],s.pitch_rate)));
  const u0=local(r0,j0),v0=Math.hypot(...u0);
  const flat=Object.values(E.aero.body_flat_plate_areas_m2).reduce((a,b)=>a+b,0);
  bodyDrag=.5*rho*v0*v0*flat;
  if(v0>1e-12){
    apply(mul(u0,-bodyDrag/v0),j0,[-r0[1],r0[0],0]);
    loss+=bodyDrag*v0;
  }
  const rT=sub([-E.coupled.tail_arm_m,0,0],q.cg),uT=local(rT,j0);
  const speedT=Math.hypot(uT[0],uT[1]);
  if(speedT>1e-12){
    const d=[uT[0]/speedT,uT[1]/speedT,0];
    const alpha=Math.atan2(-uT[1],uT[0]);
    const CL=E.coupled.tail_lift_slope*clamp(alpha+E.coupled.tail_incidence_rad,-.35,.35);
    const CD=E.coupled.tail_drag_coefficient+.08*alpha*alpha;
    const pressure=.5*rho*speedT*speedT*E.coupled.tail_area_m2;
    const F=mul(sub(mul(cross([0,0,1],d),CL),mul(d,CD)),pressure);
    apply(F,j0,[-rT[1],rT[0],0]);loss+=pressure*CD*speedT;
  }
  const horizontal=Math.hypot(air[0],air[2]);
  if(horizontal>1e-12){
    const side=.5*rho*horizontal*horizontal*E.controls.rudder_coefficient*(c.rudder??0);
    apply(mul([-air[2]/horizontal,0,air[0]/horizontal],side),j0,[-r0[1],r0[0],0]);
  }
  const throttle=clamp(c.throttle??0,0,1);
  const commanded=s.fuel_kg>0&&throttle>0&&e.motor_enabled!==false&&!s.gliding;
  const on=commanded&&s.omega>=0;
  let shaftTorque=0;
  if(on){
    const target=2*Math.PI*E.engine.maximum_wingbeat_hz*throttle;
    shaftTorque=clamp(E.engine.governor_torque_gain_Nm_s*(target-s.omega),0,
      Math.min(E.engine.starting_torque_Nm,E.engine.continuous_shaft_limit_W*throttle/Math.max(.5,Math.abs(s.omega))));
  }
  const damping=e.damping_scale??1;
  const phaseDamping=E.coupled.phase_damping_Nm_s*s.omega*damping;
  const pitchDamping=E.coupled.pitch_damping_Nm_s*s.pitch_rate*damping;
  const force=add(bodyToWorld(total,s),[0,-q.mass*g,0]);
  return {
    force,lift:bodyToWorld(wing,s)[1],thrust:wing[0],drag:bodyDrag,
    power:shaftTorque*s.omega,airspeed:Math.hypot(...air),
    aero_power_W:-Qphi*s.omega,wing_dissipation_W:loss,
    mass_kg:q.mass,Qphi,Qpitch,motor_torque_Nm:shaftTorque*E.engine.transmission_efficiency,
    control_torque_Nm:E.coupled.pitch_control_torque_Nm*(c.pitch??0),
    phase_damping_Nm:phaseDamping,pitch_damping_Nm:pitchDamping,
    wing_force_body_N:wing,mechanics:q,motor_commanded:commanded
  };
}

function heightWith(s,e,q){
  let height=-Infinity;
  const lateralY=bodyToWorld([0,0,1],s)[1];
  // Support of the actual .12 m wide, lateral-axis cylindrical tyre.
  const tyreDepth=parameters.wheel_radius_m*Math.sqrt(Math.max(0,1-lateralY*lateralY))+.06*Math.abs(lateralY);
  for(const gear of parameters.gear_centres){
    const r=bodyToWorld(sub(gear,q.cg),s);
    const ground=e.surfaceHeight?e.surfaceHeight(s.position[0]+r[0],s.position[2]+r[2]):0;
    height=Math.max(height,ground+tyreDepth-r[1]);
  }
  return height;
}
export function contactHeight(s,e){
  return heightWith(s,e,mechanics(parameters,s,wingPose));
}

export function accelerations(s,c,e){
  const f=forces(s,c,e),q=f.mechanics,w=s.omega,p=s.pitch_rate;
  const R1=f.motor_torque_Nm+f.Qphi-q.Up-f.phase_damping_Nm-.5*q.Ap*w*w+.5*q.Cp*p*p;
  const R2=f.Qpitch+f.control_torque_Nm-f.pitch_damping_Nm-q.Bp*w*w-q.Cp*w*p;
  let phase_accel,pitch_accel,grounded=false,normal_load_N=0;
  if(e.pitch_locked){
    phase_accel=R1/q.A;pitch_accel=0;
  }else{
    const determinant=q.A*q.C-q.B*q.B;
    phase_accel=(q.C*R1-q.B*R2)/determinant;
    pitch_accel=(q.A*R2-q.B*R1)/determinant;
    // Pitch=0 three-wheel holonomic reduction. Release only when normal < 0,
    // not when an instantaneous wing-lift threshold happens to be exceeded.
    const near=s.position[1]<=heightWith(s,e,q)+.0005;
    if(near&&Math.abs(s.pitch)<1e-7&&Math.abs(s.roll)<1e-7){
      const Ag=q.A+q.mass*q.dc[1]**2;
      const Agp=q.Ap+2*q.mass*q.dc[1]*q.ddc[1];
      const ga=(f.motor_torque_Nm+f.Qphi-q.Up-f.phase_damping_Nm-.5*Agp*w*w+f.force[1]*q.dc[1])/Ag;
      const N=q.mass*(q.ddc[1]*w*w+q.dc[1]*ga)-f.force[1];
      if(N>=0){
        grounded=true;normal_load_N=N;phase_accel=ga;pitch_accel=0;
        f.force=f.force.slice();f.force[1]+=N;
        const speed=Math.hypot(s.velocity[0],s.velocity[2]);
        if(speed>1e-9){
          const resistance=E.controls.rolling_friction_coefficient*N+E.controls.brake_force_N*clamp(c.brake??0,0,1);
          const friction=Math.min(resistance,speed*q.mass/E.coupled.integration_dt_max);
          f.force[0]-=friction*s.velocity[0]/speed;f.force[2]-=friction*s.velocity[2]/speed;
          f.ground_loss_W=friction*speed;
        }
      }
    }
  }
  return {...f,phase_accel,pitch_accel,grounded,normal_load_N};
}

function derivative(s,c,e){
  const a=accelerations(s,c,e);
  const rollAccel=E.controls.roll_accel_rad_s2*(c.bank??0)-E.controls.roll_damping_per_s*s.roll_rate;
  const velocity=a.grounded?[s.velocity[0],a.mechanics.dc[1]*s.omega,s.velocity[2]]:s.velocity;
  return {a,position:velocity,velocity:mul(a.force,1/a.mass_kg),
    phase:s.omega,omega:a.phase_accel,pitch:e.pitch_locked||a.grounded?0:s.pitch_rate,
    pitch_rate:a.pitch_accel,roll:a.grounded?0:s.roll_rate,roll_rate:a.grounded?0:rollAccel,
    heading:(c.rudder??0)*E.controls.rudder_coefficient,
    fuel_kg:a.motor_commanded?-(a.power*E.fuel.bsfc_kg_kWh/3600000+E.fuel.idle_kg_s):0,
    shaft_work_J:a.power,aero_work_J:a.wing_dissipation_W,
    drive_heat_J:a.power*(1-E.engine.transmission_efficiency)+a.phase_damping_Nm*s.omega+a.pitch_damping_Nm*s.pitch_rate+(a.ground_loss_W??0),
    control_work_J:a.control_torque_Nm*s.pitch_rate};
}
function advance(s,d,h){
  const n={...s,position:add(s.position,mul(d.position,h)),velocity:add(s.velocity,mul(d.velocity,h))};
  for(const key of ['phase','omega','pitch','pitch_rate','roll','roll_rate','heading','fuel_kg','shaft_work_J','aero_work_J','drive_heat_J','control_work_J'])
    n[key]=(s[key]??0)+h*d[key];
  n.fuel_kg=Math.max(0,n.fuel_kg);
  if(d.a.grounded){
    n.pitch=0;n.pitch_rate=0;n.roll=0;n.roll_rate=0;
    const q=mechanics(parameters,n,wingPose);
    n.position[1]=heightWith(n,d.environment??{},q);
    n.velocity[1]=q.dc[1]*n.omega;
  }
  return n;
}
export function step(state,c,e,dt){
  let s={...state,position:state.position.slice(),velocity:state.velocity.slice()};
  if(s.status==='crashed'||!(dt>0))return s;
  const count=Math.ceil(dt/E.coupled.integration_dt_max),h=dt/count;
  for(let i=0;i<count;i++){
    const k1=derivative(s,c,e);k1.environment=e;
    const mid=advance(s,k1,h/2);
    const k2=derivative(mid,c,e);k2.environment=e;
    const n=advance(s,k2,h);
    n.time=(s.time??0)+h;
    // There is no mode switch when the motor is cut: the same mechanism coasts.
    const q=mechanics(parameters,n,wingPose),height=heightWith(n,e,q);
    if(n.position[1]<height&&!e.pitch_locked){
      let bodySink=Infinity;
      for(const gear of parameters.gear_centres){
        const r=sub(gear,q.cg);
        const rotational=add(mul([-r[1],r[0],0],n.pitch_rate),
          mul([0,-r[2],r[1]],n.roll_rate));
        const relative=bodyToWorld(sub(rotational,mul(q.dc,n.omega)),n);
        bodySink=Math.min(bodySink,n.velocity[1]+relative[1]);
      }
      const unsafe=bodySink < -E.controls.safe_touchdown_sink_m_s||
        Math.hypot(n.velocity[0],n.velocity[2])>E.controls.safe_touchdown_speed_m_s||
        Math.abs(n.roll)>E.controls.safe_touchdown_bank_rad;
      if(!k1.a.grounded&&unsafe)n.status='crashed';
      else{
        n.pitch=0;n.pitch_rate=0;n.roll=0;n.roll_rate=0;
        n.position[1]=heightWith(n,e,q);n.velocity[1]=q.dc[1]*n.omega;n.status='grounded';
      }
    }else n.status=k2.a.grounded?'grounded':'flying';
    // Wheel geometry is built in the XY plane: rolling is around lateral Z.
    const forward=worldToBody(n.velocity,n)[0];
    n.wheel_angle=(s.wheel_angle??0)-forward*h/parameters.wheel_radius_m;
    n.roll=clamp(n.roll,-E.controls.bank_limit_rad,E.controls.bank_limit_rad);
    s=n;
    if(s.status==='crashed')break;
  }
  return s;
}
